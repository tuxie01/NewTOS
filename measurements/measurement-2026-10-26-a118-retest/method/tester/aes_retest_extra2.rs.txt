
/// Same order check for a wind_update(BEG) waiter (opcode 107 blocks inside its body, not in the crysbind gate).
#[test]
fn r_wind_update_re_execution_keeps_the_wait_line_order() {
    let (_g, _host) = lock();
    let mut p = pair();
    let c = Proc::new(true);
    let mut pc = Aespb::new();
    c.run();
    appl_init(&mut pc);
    sched::unready(c.slot);
    p.a_holds();
    assert_eq!(wind_update(&mut p.pbb, BEG_UPDATE).ret, None, "B waits behind A");
    c.run();
    assert_eq!(wind_update(&mut pc, BEG_UPDATE).ret, None, "C waits behind A and B");
    p.a.run();
    assert_eq!(wind_update(&mut p.pa, END_UPDATE).ret, Some(1));
    p.b.run();
    assert_eq!(p.pbb.trap().ret, Some(1), "B's BEG_UPDATE completes");
    assert_eq!(test_update_lock(), (1, Some(p.b.slot)), "B holds it (C still waits)");
}
