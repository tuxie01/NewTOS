// ── TESTER: state changes while parked ────────────────────────────────────
/// B parks on wind_set(h); the holder A deletes h; on hand-over B's re-executed call refuses.
/// Expectation (EmuTOS: the lock is B's, B ends it): lock must end (0,None).
#[test]
fn t_wind_set_parked_then_window_deleted_does_not_leak_lock() {
    let (_g, _host) = lock();
    let mut p = pair();
    p.b.run();
    let h = wind_create(&mut p.pbb);
    p.a_holds();
    let o = wind_set(&mut p.pbb, h, WF_NAME);
    p.assert_parked(&o, "wind_set");
    p.a.run();
    assert_eq!(wind_delete(&mut p.pa, h).ret, Some(1));
    assert_eq!(wind_update(&mut p.pa, END_UPDATE).ret, Some(1));
    p.b.run();
    let r = p.pbb.trap();
    eprintln!("B re-exec ret={:?} lock={:?}", r.ret, test_update_lock());
    assert_eq!(test_update_lock(), (0, None), "lock leaked to B after refusal on re-execution");
}

/// B parks on wind_close(h); A closes h.
