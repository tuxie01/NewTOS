# NewTOS / PiRusTOS — Raspberry Pi 4 (BCM2711) firmware boot config.
#
# Copy this file, `kernel8.img`, and the Pi firmware files onto a
# FAT32 boot partition. See docs/HW-TEST-LOG.md for the full SD-card
# layout and the USB-TTL wiring.

# Boot a 64-bit (AArch64) bare-metal kernel image.
arm_64bit=1
kernel=kernel8.img

# ── HDMI display ────────────────────────────────────────────────────
# Force a known HDMI mode so the VideoCore framebuffer comes up headless
# (DMT 1080p60 = group 2 / mode 82). hotplug forces output even with no
# monitor attached at boot.
hdmi_group=2
hdmi_mode=82
hdmi_force_hotplug=1
hdmi_pixel_encoding=2

# ── GPU memory split (CRITICAL for the framebuffer) ─────────────────
# Reserve 256 MiB for the VideoCore. WITHOUT this the firmware gives the
# GPU minimal memory and the VDI/HDMI framebuffer is unavailable
# ("[fb] VideoCore mailbox framebuffer unavailable — no graphics"); the
# ARM side then also sees ~977 MiB instead of ~768 MiB.
gpu_mem=256

# Disable unused peripherals (kept from the verified real-HW config).
dtparam=audio=off
dtparam=i2c_arm=off
dtparam=spi=off

# NOTE on the GIC-400 RX IRQ: a self-built armstub8-gic.bin was tried to
# move the GIC SPIs to Group 1, but it did not help (IGROUPR stayed 0 on
# this firmware). Console input uses the UART0 poll workaround instead;
# the serial console is dev-only anyway. See docs/MEMORY.md QUIRK-002.
# (The earlier "armstub broke the framebuffer" note was wrong — that was
# a config.txt clobber that dropped gpu_mem, fixed above.)

# ── Serial console (PL011 UART0 on GPIO 14/15) ──────────────────────
# Enable the primary UART (PL011 / UART0). On the Pi 4 this routes to
# GPIO 14 (TXD) / 15 (RXD) — the header pins the kernel's bcm2711
# driver configures as ALT0.
enable_uart=1
# ── DIAGNOSE, wieder entfernen ──────────────────────────────────────
# Legt die Ausgabe der zweiten Firmware-Bootstufe auf die serielle
# Leitung. Zweck: trennen, ob Adapter, Verkabelung, Terminal und
# Baudrate funktionieren, bevor unser Kernel überhaupt startet.
#   Ausgabe erscheint  → serielle Kette in Ordnung, Fehler liegt bei uns
#   weiterhin nichts   → das Problem liegt vor unserem Kernel
uart_2ndstage=1


# CRITICAL: pin the UART reference clock to 48 MHz. The kernel uses a
# fixed baud divisor (IBRD=26, FBRD=3 → 115200 baud) that assumes a
# 48 MHz UART clock. Without this the firmware lets the UART clock
# track the (variable) core frequency and the baud rate drifts —
# you get garbage or nothing on the serial line. This single line is
# the classic Pi 4 bare-metal serial gotcha.
init_uart_clock=48000000

# Keep the CPU clock from idling the core down in a way that could
# still perturb peripheral clocks during early bring-up.
force_turbo=0

# The firmware passes a device-tree-blob pointer in x0; the kernel
# parses it for the physical memory map (falling back to a compiled-in
# 1 GiB if absent). If DTB parsing ever misbehaves on real hardware,
# uncomment the next line to boot with no DTB (x0 = 0, compiled-in
# memory size):
#device_tree=
